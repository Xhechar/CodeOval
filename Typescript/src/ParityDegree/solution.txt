export function solution(N: number): number {
  let count = 0;

  while (N % 2 === 0) {
    count++;
    N = N / 2;
  }

  return count;
}

export function solution(N: number): number {
  let count = 0;

  while ((N & 1) === 0) {
    count++;
    N >>= 1;
  }

  return count;
}

N & 1 checks the last bit.

If last bit is 0 → number is even.

Shift right removes that zero.

This is still O(log N), but lower-level and elegant.

Even more nerdy: in some languages there’s a built-in “count trailing zeros” instruction at the CPU level.

This problem is secretly about understanding binary representation.
